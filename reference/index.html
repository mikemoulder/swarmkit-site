import type { Allocation, Bid, Demand, RoundSpec } from "./types.ts";

export type { Allocation, Bid, Demand, RoundSpec } from "./types.ts";
export type RoundCommand = {
    type: "demand";
    buyerId: string;
    request: Demand | null;
} | {
    type: "commitment";
    sellerId: string;
    hash: string;
} | {
    type: "reveal";
    sellerId: string;
    bid: Bid;
};
export type RoundEvent = RoundCommand & {
    at: number;
};
export type RoundTranscript = {
    version: 1;
    spec: RoundSpec;
    events: readonly RoundEvent[];
    observedAt: number;
};
export declare function commitBid(roundId: string, sellerId: string, bid: Bid): Promise<string>;
export declare function createRound(input: RoundSpec, clock?: () => number, mechanism?: ProcurementMechanism): {
    submitDemand: (buyerId: string, request: Demand | null) => boolean;
    submitCommitment: (sellerId: string, hash: string) => boolean;
    revealBid: (sellerId: string, bid: Bid) => Promise<boolean>;
    snapshot: () => {
        splitAllocation?: SplitAllocation | null | undefined;
        spec: RoundSpec;
        phase: string;
        buyers: {
            quantity: string;
            budget: string;
            buyerId: string;
        }[];
        quantity: string;
        bids: {
            sellerId: string;
            commitment: string;
            reveal: {
                capacity: string;
                unitPrice: string;
                salt: string;
            } | null;
            rejection: string | null;
        }[];
        allocation: Allocation | null;
    };
    transcript: () => RoundTranscript;
};
export type RoundSnapshot = ReturnType<ReturnType<typeof createRound>["snapshot"]>;

import type { RoundSnapshot, RoundTranscript } from "./round.ts";
import type { Bid } from "./types.ts";
import type { SimulationOptions } from "./simulation.ts";
/** Private simulation checkpoint. Contains unrevealed offers; never publish it. */
export type SimulationSession = {
    version: 1;
    stage: "ready" | "demand" | "commit" | "reveal" | "closed";
    randomState: number;
    transcript: RoundTranscript;
    sealedOffers: {
        sellerId: string;
        bid: Bid;
        reveal: boolean;
    }[];
};
/** Start an empty simulation. No strategies execute until advanceSimulation is called. */
export declare function startSimulation(options: Pick<SimulationOptions, "spec" | "seed" | "buyers" | "sellers" | "mechanism">): SimulationSession;
/** Advance exactly one stage. Persist the returned checkpoint before calling again.
 * Strategies must be side-effect-free: an uncommitted attempt may be retried.
 * The supplied callbacks may be asynchronous; clocks are logical simulation time.
 */
export declare function advanceSimulation(input: SimulationSession, strategies: Pick<SimulationOptions, "buyers" | "sellers" | "mechanism">): Promise<SimulationSession>;
/** Explain the recorded outcome without claiming the heuristic proves infeasibility. */
export declare function explainAllocation(snapshot: RoundSnapshot): {
    outcome: "allocated" | "no-sale";
    reason: string;
    buyers: {
        spent: string;
        remainingBudget: string;
        lines: (import("./mechanism.ts").FulfillmentLine & {
            unitPrice: string;
            amount: string;
        })[];
        quantity: string;
        budget: string;
        buyerId: string;
    }[];
    sellers: {
        sellerId: string;
        reason: string;
        quantity: string;
    }[];
};
